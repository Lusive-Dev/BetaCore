cmake_minimum_required(VERSION 3.16)
project(NexiumRuntime VERSION 0.3.0 LANGUAGES CXX)
option(NEXIUM_RUNTIME_BUILD_TESTS "Build runtime tests" ON)
find_package(Threads REQUIRED)
add_library(nexium_runtime STATIC runtime/src/runtime.cpp runtime/src/platform.cpp runtime/src/scheduler.cpp runtime/src/abi.cpp)
target_include_directories(nexium_runtime PUBLIC runtime/include)
target_compile_features(nexium_runtime PUBLIC cxx_std_17)
target_link_libraries(nexium_runtime PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(nexium_runtime PUBLIC ws2_32 bcrypt)
endif()
set_target_properties(nexium_runtime PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
add_library(nexium_frontend STATIC compiler/src/lexer.cpp compiler/src/parser.cpp compiler/src/semantic.cpp)
target_include_directories(nexium_frontend PUBLIC compiler/include)
target_compile_features(nexium_frontend PUBLIC cxx_std_17)
set_target_properties(nexium_frontend PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(nexium_frontend PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(nexium_frontend PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(MSVC)
  target_compile_options(nexium_runtime PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(nexium_runtime PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(NEXIUM_RUNTIME_BUILD_TESTS)
  enable_testing()
  add_executable(nexium_runtime_tests runtime/tests/runtime_tests.cpp)
  target_link_libraries(nexium_runtime_tests PRIVATE nexium_runtime)
  add_test(NAME nexium_runtime_tests COMMAND nexium_runtime_tests)
  add_executable(nexium_lexer_tests compiler/tests/lexer_tests.cpp)
  target_link_libraries(nexium_lexer_tests PRIVATE nexium_frontend)
  add_test(NAME nexium_lexer_tests COMMAND nexium_lexer_tests)
  add_executable(nexium_parser_tests compiler/tests/parser_tests.cpp)
  target_link_libraries(nexium_parser_tests PRIVATE nexium_frontend)
  add_test(NAME nexium_parser_tests COMMAND nexium_parser_tests)
  add_executable(nexium_semantic_tests compiler/tests/semantic_tests.cpp)
  target_link_libraries(nexium_semantic_tests PRIVATE nexium_frontend)
  add_test(NAME nexium_semantic_tests COMMAND nexium_semantic_tests)
endif()
